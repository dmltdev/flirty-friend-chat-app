  import type { UIMessage } from "ai";
  import type { ModerationViolation } from "@/app/features/moderation/types";

  export type ChatMessage = UIMessage;

  export type ChatSession = {
    id: string;
    messages: ChatMessage[];
    createdAt: number;
  };

  export type ChatError =
    | { type: "moderation_blocked"; violations: ModerationViolation[] }
    | { type: "rate_limited"; retryAfterSec: number }
    | { type: "bad_request"; message: string }
    | { type: "unknown"; message?: string };

  Notes:
  - type is the discriminant — narrowing via switch (err.type) gives you each
  variant's payload.
  - moderation_blocked carries ModerationViolation[] (matches
  ModerationResult.violations).
  - rate_limited carries retryAfterSec (matches RateLimitResult's failure shape
  at rate-limit.ts:11).
  - bad_request holds a human-readable reason (e.g. malformed payload, missing
  fields).
  - unknown is the generic catch-all for server errors / unexpected failures;
  message optional so you can omit it in production.
